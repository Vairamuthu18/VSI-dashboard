function setupCanvas(canvas) {
const dpr = window.devicePixelRatio || 1;
canvas.width = canvas.offsetWidth * dpr;
canvas.height = canvas.offsetHeight * dpr;
const ctx = canvas.getContext('2d');
ctx.scale(dpr, dpr);
return { ctx, w: canvas.offsetWidth, h: canvas.offsetHeight };
}

function fmtNum(n) {
if (n >= 1e6) return (n/1e6).toFixed(1)+'M';
if (n >= 1e3) return (n/1e3).toFixed(1)+'K';
return n.toString();
}

function fmtDate(d) {
return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
}

// Sparkline
function drawSparkline(canvas, data, color) {
const { ctx, w, h } = setupCanvas(canvas);
const max = Math.max(...data) * 1.1;
const min = Math.min(...data) * 0.9;
const range = max - min || 1;

// Fill
const grad = ctx.createLinearGradient(0, 0, 0, h);
grad.addColorStop(0, color + '28');
grad.addColorStop(1, color + '03');

ctx.beginPath();
data.forEach((v, i) => {
const x = (i / (data.length - 1)) * w;
const y = h - ((v - min) / range) * h * 0.85 - h * 0.05;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.lineTo(w, h); ctx.lineTo(0, h);
ctx.fillStyle = grad; ctx.fill();

// Line
ctx.beginPath();
data.forEach((v, i) => {
const x = (i / (data.length - 1)) * w;
const y = h - ((v - min) / range) * h * 0.85 - h * 0.05;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.strokeStyle = color; ctx.lineWidth = 2; ctx.stroke();

// End dot
const lx = w;
const ly = h - ((data[data.length-1] - min) / range) * h * 0.85 - h * 0.05;
ctx.beginPath(); ctx.arc(lx, ly, 3, 0, Math.PI*2);
ctx.fillStyle = color; ctx.fill();
}

// Main area / multi-line chart
function drawMentionsChart(canvas, series, view) {
const { ctx, w, h } = setupCanvas(canvas);
const pad = { t: 20, r: 20, b: 44, l: 50 };
const cw = w - pad.l - pad.r;
const ch = h - pad.t - pad.b;

ctx.clearRect(0, 0, w, h);

// Grid
ctx.strokeStyle = 'rgba(30,41,59,0.6)';
ctx.lineWidth = 1;
for (let i = 0; i <= 5; i++) {
const y = pad.t + (i / 5) * ch;
ctx.beginPath(); ctx.moveTo(pad.l, y); ctx.lineTo(w - pad.r, y); ctx.stroke();
}

if (view === 'combined') {
const data = series.map(s => s.total);
const max = Math.max(...data) * 1.2;

// Gradient
const grad = ctx.createLinearGradient(0, pad.t, 0, pad.t + ch);
grad.addColorStop(0, '#6366f130');
grad.addColorStop(1, '#6366f103');

ctx.beginPath();
data.forEach((v, i) => {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - v / max) * ch;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.lineTo(pad.l + cw, pad.t + ch); ctx.lineTo(pad.l, pad.t + ch);
ctx.fillStyle = grad; ctx.fill();

ctx.beginPath();
data.forEach((v, i) => {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - v / max) * ch;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.strokeStyle = '#6366f1'; ctx.lineWidth = 2.5; ctx.stroke();

// Dots
data.forEach((v, i) => {
if (i % 4 === 0 || i === data.length - 1) {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - v / max) * ch;
ctx.beginPath(); ctx.arc(x, y, 3.5, 0, Math.PI*2);
ctx.fillStyle = '#6366f1'; ctx.fill();
}
});

// Y labels
ctx.fillStyle = '#64748b'; ctx.font = '11px Inter'; ctx.textAlign = 'right';
for (let i = 0; i <= 5; i++) {
const val = Math.floor(max * (1 - i / 5));
ctx.fillText(val, pad.l - 8, pad.t + (i / 5) * ch + 4);
}
} else {
const keys = ['chatgpt', 'gemini', 'perplexity', 'claude', 'copilot', 'meta'];
const colors = ['#10a37f', '#4285f4', '#20b2aa', '#cc785c', '#7b61ff', '#0668E1'];
let allMax = 0;
keys.forEach(k => {
const m = Math.max(...series.map(s => s[k]));
if (m > allMax) allMax = m;
});
allMax *= 1.2;

keys.forEach((k, ki) => {
const data = series.map(s => s[k]);
ctx.beginPath();
data.forEach((v, i) => {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - v / allMax) * ch;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.strokeStyle = colors[ki]; ctx.lineWidth = 2; ctx.stroke();
});

// Y labels
ctx.fillStyle = '#64748b'; ctx.font = '11px Inter'; ctx.textAlign = 'right';
for (let i = 0; i <= 5; i++) {
ctx.fillText(Math.floor(allMax * (1 - i / 5)), pad.l - 8, pad.t + (i / 5) * ch + 4);
}

// Legend
ctx.textAlign = 'left'; ctx.font = '10px Inter';
keys.forEach((k, ki) => {
const lx = pad.l + ki * 90;
const ly = h - 6;
ctx.fillStyle = colors[ki];
ctx.fillRect(lx, ly - 4, 14, 3);
ctx.fillStyle = '#94a3b8';
ctx.fillText(k.charAt(0).toUpperCase() + k.slice(1), lx + 18, ly);
});
}

// X labels
ctx.fillStyle = '#64748b'; ctx.font = '10px Inter'; ctx.textAlign = 'center';
const step = Math.max(1, Math.floor(series.length / 7));
series.forEach((s, i) => {
if (i % step === 0 || i === series.length - 1) {
const x = pad.l + (i / (series.length - 1)) * cw;
ctx.fillText(fmtDate(s.date), x, h - pad.b + 18);
}
});

// Hover
canvas.onmousemove = function(e) {
const rect = canvas.getBoundingClientRect();
const mx = e.clientX - rect.left;
const idx = Math.round(((mx - pad.l) / cw) * (series.length - 1));
if (idx >= 0 && idx < series.length && mx >= pad.l && mx <= w - pad.r) {
const tip = document.getElementById('tip');
const s = series[idx];
tip.innerHTML = `<strong>${fmtDate(s.date)}</strong><br>Total: ${s.total}<br>ChatGPT: ${s.chatgpt} · Gemini: ${s.gemini}<br>Perplexity: ${s.perplexity} · Claude: ${s.claude}`;
tip.style.left = (e.clientX + 16) + 'px';
tip.style.top = (e.clientY - 10) + 'px';
tip.classList.add('show');
} else {
document.getElementById('tip').classList.remove('show');
}
};
canvas.onmouseleave = () => document.getElementById('tip').classList.remove('show');
}

// Category donut
function drawCategoryChart(canvas, categories) {
const { ctx, w, h } = setupCanvas(canvas);
const cx = w * 0.35, cy = h * 0.5, r = Math.min(w, h) * 0.32;

let start = -Math.PI / 2;
categories.forEach(cat => {
const slice = (cat.pct / 100) * Math.PI * 2;
ctx.beginPath();
ctx.arc(cx, cy, r, start, start + slice);
ctx.arc(cx, cy, r * 0.55, start + slice, start, true);
ctx.closePath();
ctx.fillStyle = cat.color;
ctx.fill();
start += slice;
});

// Center text
ctx.fillStyle = '#f1f5f9'; ctx.font = 'bold 22px Inter'; ctx.textAlign = 'center';
ctx.fillText('100%', cx, cy + 3);
ctx.fillStyle = '#64748b'; ctx.font = '10px Inter';
ctx.fillText('Coverage', cx, cy + 18);

// Legend
ctx.textAlign = 'left';
categories.forEach((cat, i) => {
const lx = w * 0.68;
const ly = 30 + i * 38;
ctx.fillStyle = cat.color;
ctx.beginPath(); ctx.roundRect(lx, ly - 6, 12, 12, 3); ctx.fill();
ctx.fillStyle = '#e2e8f0'; ctx.font = '12px Inter';
ctx.fillText(cat.name, lx + 18, ly + 4);
ctx.fillStyle = '#94a3b8'; ctx.font = 'bold 12px Inter';
ctx.fillText(cat.pct + '%', lx + 18, ly + 20);
});
}

const keys = ['chatgpt', 'gemini', 'perplexity', 'claude', 'copilot', 'meta'];

const grad = ctx.createLinearGradient(0, pad.t, 0, pad.t + ch);
grad.addColorStop(0, color + '35');
grad.addColorStop(1, color + '08');

ctx.beginPath();
data.forEach((d, i) => {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - d[k] / 45) * ch;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});

const lastX = pad.l + cw;
ctx.lineTo(lastX, pad.t + ch);
ctx.lineTo(pad.l, pad.t + ch);
ctx.fillStyle = grad; ctx.fill();

ctx.beginPath();
data.forEach((d, i) => {
const x = pad.l + (i / (data.length - 1)) * cw;
const y = pad.t + (1 - d[k] / 45) * ch;
i === 0 ? ctx.moveTo(x, y) : ctx.lineTo(x, y);
});
ctx.strokeStyle = color; ctx.lineWidth = 2; ctx.stroke();
});
keys.reverse();

// X labels
ctx.fillStyle = '#64748b'; ctx.font = '10px Inter'; ctx.textAlign = 'center';
data.forEach((d, i) => {
if (i % 2 === 0 || i === data.length - 1) {
const x = pad.l + (i / (data.length - 1)) * cw;
ctx.fillText(fmtDate(d.date), x, h - pad.b + 18);
}
});

// Y labels
ctx.textAlign = 'right';
for (let i = 0; i <= 4; i++) {
ctx.fillText(Math.floor(45 * (1 - i / 4)) + '%', pad.l - 8, pad.t + (i / 4) * ch + 4);
}

// Legend
ctx.textAlign = 'left'; ctx.font = '10px Inter';
keys.forEach((k, ki) => {
const lx = pad.l + ki * 100;
ctx.fillStyle = colors[ki];
ctx.fillRect(lx, h - 8, 14, 3);
ctx.fillStyle = '#94a3b8';
ctx.fillText(k.charAt(0).toUpperCase() + k.slice(1), lx + 18, h - 4);
});
}

renderKeywords(data.keywords);

}
